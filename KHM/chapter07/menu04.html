02កម្មវិធីថែរក្សាសុខភាពនៅមណ្ឌលសុខភាព

ជាកម្មវិធីដែលផ្តល់សេវាកម្មគ្រប់គ្រងសុខភាពដូចជា ការគ្រប់គ្រងជំងឺរ៉ាំរ៉ៃដែលមានអ្នកជំនាញសុខភាពរបស់មណ្ឌសុខភាពដូចជា គិលានុបដ្ឋាយិកា អ្នកព្យាបាលរាងកាយ និងអ្នកជំនាញអាហារូបត្ថម្ភ ទៅដល់ផ្ទះផ្ទាល់ ដោយផ្តោតទៅលើក្រុមងាយរងគ្រោះផ្នែកសង្គមវប្បធម៌ និងសេដ្ឋកិច្ច និងមនុស្សចាស់ រស់នៅតែម្នាក់ឯង ដែលមានអាយុលើស ៦៥ឆ្នាំ និងគូស្វាមីភរិយាវ័យចំណាស់ដែលមានអាយុលើស ៧៥ឆ្នាំ ដែលមានការលំបាក ក្នុងការប្រើប្រាស់សេវាថែទាំសុខភាព។

របៀបប្រើប្រាស់សេវាកម្ម
  • ទៅមជ្ឈមណ្ឌលសុខុមាលភាពសុខភាពផ្លូវចិត្ត និងរៀបចំពាក្យចុះឈ្មោះ។
  • ករណីដែលអ្នកជំងឺទៅមជ្ឈមណ្ឌលសុខុមាលភាពសុខភាពផ្លូវចិត្ត ជាលើកដំបូង គេត្រូវធ្វើការប្រឹក្សា និងតេស្តចិត្តសាស្រ្ត។ ហើយអ្នកជំងឺអាចទទួល ប្រើប្រាស់សេវាកម្មទៅតាមលទ្ធផលប្រឹក្សា និងតេស្ត ។
  • សំរាប់ព័ត៌មានទាក់ទងជាមួយនឹងមជ្ឈមណ្ឌលសុខុមាលភាពសុខភាពផ្លូវចិត្ត សូមទូរស័ព្ទទៅកាន់លេខ១៥៧៧-០១៩៩ របស់មជ្ឈមណ្ឌលប្រឹក្សាសុខភាពផ្លូវចិត្ត ។
តំលៃសេវាកម្ម :
មិនគិតថ្លៃ(គិតថ្លៃក្នុងករណីដែលអ្នកជំងឺប្រើប្រាស់ កម្មវិធីដែលត្រូវការការព្យបាលដែលមានតំលៃខ្ពស់)។
