ຄົນຕ່າງປະເທດທີ່ຕົກຢູ່ພາຍໃຕ້ໝວດຕໍ່ໄປນີ້ ແລະ ມີເງື່ອນໄຂເໝາະສົມກັບໝວດທີ່ສາມາດນໍາໃຊ້ໄດ້ ສາມາດສະໝັກຂໍສັນຊາດໄດ້ຢູ່ທີ່ຫ້ອງການຄົນເຂົ້າເມືອງ 19 ແຫ່ງ ຫຼື ຫ້ອງການຂອງລັດຖະບານຕ່າງປະເທດ ທີ່ຢູ່ໃນຂອບເຂດຄວາມຮັບຜິດຊອບຕໍ່ວຽກງານສັນຊາດ. ຜູ້ສະໝັກຕ້ອງກຽມເອກະສານທີ່ຈໍາເປັນທັງໝົດໂດຍອ້າງອີງໃສ່ ຂໍ້ມູນການແລກປ່ຽນຂໍ້ມູນ (ສັນຊາດ/ການປ່ຽນສັນຊາດ) ຂອງໜ້າທໍາອິດຂອງລັດຖະບານອີເລັກໂທຣນິກ ສໍາລັບຄົນຕ່າງປະເທດ (www.hikorea.go.kr) ກັບຫ້ອງການຄົນເຂົ້າເມືອງຂອງເຂດອໍານາດຕັດສິນ.

01ເງື່ອນໄຂ

  1. 1ຊາວຕ່າງຊາດ ທີ່ມີອາຍຸຄົບກະສຽນຕາມທີ່ກຳນົດໄວ້ໃນ 「ກົດໝາຍແພ່ງ」ຂອງເກົາຫຼີ ໂດຍທີ່ໄດ້ພັກເຊົາຖາວອນ ໃນເກົາຫຼີ ເປັນເວລາ 5 ປີ ຕໍ່ເນື່ອງກັນ ແລະ ມີທີ່ຢູ່ທີ່ໝັ້ນຄົງ ຢູ່ໃນເກົາຫຼີ.
  2. 2ຄົນຕ່າງປະເທດທີ່ພໍ່ ຫຼື ແມ່ເປັນ/ເຄີຍເປັນຄົນສັນຊາດເກົາຫຼີ; ຄົນຕ່າງປະເທດທີ່ເກີດຢູ່ໃນເກົາຫຼີ ແລະ ພໍ່ຫຼືແມ່ເກີດຢູ່ເກົາຫຼີ; ແລະ ຄົນຕ່າງປະເທດທີ່ໄດ້ຮັບສັນຊາດເປັນຄົນເກົາຫຼີ ຫຼັງຈາກທີ່ເຂົາເຈົ້າມີອາຍຸຄົບຕາມກົດໝາຍ
  3. 3ຄົນຕ່າງປະເທດ ທີ່ໄດ້ມາແຕ່ງດອງກັບຄົນເກົາຫຼີ ຫຼື ເຄີຍແຕ່ງງານກັບຄົນເກົາຫຼີ ມາກ່ອນ(ລາຍລະອຽດ ແມ່ນໃຫ້ອ່ານເບິ່ງຄຳ ແນະນຳ ກ່ຽວກັບການຂໍສັນຊາດເກົາຫຼີຂອງຄົນຕ່າງປະເທດ ທີ່ເຂົ້າເມືອງເກົາຫຼີຖາວອນ).
  4. 4ຄົນຕ່າງຊາດຜູ້ທີ່ພໍ່ ຫຼືແມ່ ເປັນພົນລະເມືອງເກົາຫຼີ.
    • ໃນກໍລະນີ ພໍ່ ຫຼື ແມ່ ເປັນຄົນຕ່າງປະເທດ ທີ່ໄດ້ຕັດສັນຊາດ ມາເປັນຄົນເກົາຫຼີ, ບຸກຄົນດັ່ງກ່າວ ສາມາດສະເໜີຂໍສັນຊາດເເກົາຫຼີໄດ້ ໂດຍບໍ່ກ່ຽວຂ້ອງກັບອາຍຸ, ການແຕ່ງງານກັບຄົນເກົາຫຼີ ແລະ ໄລຍະເວລາການພັກເຊົາໃນເກົາຫຼີ(ແຕ່ຍົກເວັ້ນ ບຸກຄົນ ທີ່ມາເປັນລູກລ້ຽງ ຕອນທີ່ອາຍຸເຕັມກະສຽນແລ້ວ)
  5. 5ຄົນຕ່າງປະເທດ ທີ່ໄດ້ສ້າງຜົນງານຄວາມດີພິເສດ ຕໍ່ປະເທດເກົາຫຼີ
  6. 6ຄົນຕ່າງປະເທດ ທີ່ມີຄວາມສາມາດໂດດເດັ່ນ ແລະ ປະເມີນໄດ້ວ່າ ຈະສາມາດປະກອບສ່ວນເຂົ້າໃນການພັດທະນາສ້າງສາ ຄວາມຈະເລີນຮຸ່ງເຮືອງ ແກ່ປະເທດເກົາຫຼີ ໃນອະນາຄົດ.
ເງື່ອນໄຂການຂໍສັນຊາດຂອງຄົນເຂົ້າເມືອງຖາວອນ ເພື່ອແຕ່ງງານ
  1. 1ຄົນຕ່າງຊາດ ທີ່ພັກເຊົາໃນເກົາຫຼີດົນກວ່າ 2 ປີ ໃນສະຖານະແຕ່ງງານກັບພົນລະເມືອງເກົາຫຼີ
  2. 2ຄົນຕ່າງຊາດ ຜູ້ທີ່ແຕ່ງງານກັບພົນລະເມືອງເກົາຫຼີ ເປັນເວລາ 3 ປີຂື້ນໄປ ແລະໄດ້ພັກເຊົາ ຢູ່ໃນເກົາຫຼີເປັນເວລາ 1ປີ ຢ່າງຕໍ່ເນື່ອງ ໃນຖານະແຕ່ງງານກັບຄົນເກົາຫຼີ.
  3. 3ຄົນຕ່າງປະເທດ ທີ່ແຕ່ງງານກັບຄົນເກົາຫຼີ, ແຕ່ຄູ່ສົມລົດຄົນເກົາຫຼີ ໄດ້ເສຍຊີວິດ, ຫາຍສາບສູນ ຫຼື ເນື່ອງຈາກເຫດການອື່ນໆ ທີ່ເຮັດໃຫ້ ທັງສອງ ບໍ່ສາມາດຮ່ວມຊີວິດຜົວເປແປກມຍໄດ້ ໂດຍທີ່ໄດ້ພັກເຊົາຕາມໄລຍະເວລາ ທີ່ກຳນົດມາ ໃນຂໍ້①ຫຼື②.
  4. 4ຄົນຕ່າງປະເທດ ທີ່ໄດ້ແຕ່ງງານກັບຄົນເກົາຫຼີແລ້ວ ມີລູກ ທີ່ມີອາຍຸຍັງບໍ່ຄົບກະສຽນ ໂດຍທີ່ຜູ້ກ່ຽວ ຕ້ອງໄດ້ລ້ຽງດູ ຫຼື ມີພັນທະ ໃນການລ້ຽງດູລູກ ແລະ ມີເງື່ອນໄຂພັກເຊົາຢູ່ເກົາຫຼີ ຕາມໄລຍະເວລາທີ່ກຳນົດມາໃນຂໍ້ ① ຫຼື ② ເທິງນີ້.

02ເງື່ອນໄຂ

  1. 1ຕ້ອງເປັນຄົນທີ່ ເຄົາລົບຕໍ່ລະບຽບກົດໝາຍ ແລະ ມີຄຸນສົມບັດສຸພາບຮຽບຮ້ອຍ ຕາມເງື່ອນໄຂ ທີ່ກຳນົດໄວ້ໃນລະບຽບ ການຂອງ ກະຊວງ ຍຸຕິທຳ.
  2. 2ຕ້ອງເປັນຄົນ ທີ່ມີຄວາມສາມາດໃນການຫາລ້ຽງຊີບ ດ້ວຍຊັບສິນ ຫຼື ຄວາມຮູ້ຂອງຄົນເອງ ຫຼື ມີຄອບຄົວ ທີ່ສາມາດລ້ຽງດູ ຕົນເອງໄດ້.
  3. 3ຕ້ອງເປັນຄົນ ທີ່ມີຄວາມເຂົ້າໃຈຕໍ່ພາສາເກົາຫຼີ ແລະ ຮິດຄອງປະເພນີເກົາຫຼີ ເປັນຢ່າງດີ ແລະ ມີຄວາມຮູ້ພື້ນຖານດ້ານສັງຄົມ ທີ່ເໝາະສົມກັບພົນລະເມືອງເກົາຫຼີ.
  4. 4ການອະນຸຍາດໃຫ້ສັນຊາດເກົາຫຼີ ແກ່ບຸກຄົນຕ່າງປະເທດໃດໜຶ່ງ ສະແດງໃຫ້ເຫັນວ່າ ລັດຖະມົນຕີກະຊວງຍຸຕິທຳ ສາມາດຍອມ ຮັບວ່າ ຜູ້ກ່ຽວ ປະກອບສ່ວນເຂົ້າໃນການປ້ອງກັນຄວາມປອດໄພແຫ່ງຊາດ ແລະ ເຄົາລົບລະບຽບກົດໝາຍ ແລະ ບໍ່ມີຜົນກະທົບ ຕໍ່ສະ ຫວັດດີການສາທາລະນະ.

03ເອກະສານທີ່ຈໍາເປັນສໍາລັບການສະໝັກຂໍອຳນວຍຄວາມສະດວກການປ່ຽນສັນຊາດຕາມ (ແຕ່ງງານ)

  • ສໍາລັບເອກະສານທີ່ຈໍາເປັນໃນເວລາສະໝັກຂໍອຳນວຍຄວາມສະດວກການປ່ຽນສັນຊາດຕາມ (ແຕ່ງງານ), ອ້າງອີງໃສ່ ມຸມການແລກປ່ຽນຂໍ້ມູນ (ສັນຊາດ/ການປ່ຽນສັນຊາດ) ຂອງເວັບໄຊທລັດຖະບານອີເລັກໂທຣນິກ (www.hikorea.go.kr)
ຄຳຮ້ອງສະໝັກຂໍອະນຸຍາດປ່ຽນສັນຊາດ (ຕິດຮູບສີ 1 ແຜ່ນຂະໜາດ 3.5x4.5cm, ດາວໂຫຼດໄດ້ຈາກໜ້າຫຼັກ)
ສຳເນົາໜັງສືຜ່ານແດນ, ສຳເນົາບັດປະຈຳຕົວຕົ້ນສະບັບຈາກປະເທດບ້ານເກີດ (ປະກອບມີການລົງທະບຽນຜູ້ຢູ່ໃນກໍລະນີເປັນຄົນຈີນ)
  • ຕ້ອງສະເໜີຕົ້ນສະບັບເມື່ອສະໝັກ
ໃບຢັ້ງຢືນການບັນທຶກຄະດີອາຍາກັບ ອັກຄະສາວົກ ທີ່ອອກໃຫ້ໂດຍລັດຖະບານຂອງປະເທດຕົ້ນກໍາເນີດ (ປະເທດທີສາມ) ຫຼືການຢືນຢັນກົງສຸນຈາກສະຖານທູດເກົາຫຼີໃນປະເທດທີ່ອອກໃຫ້)
ສຳເນົາ 1 ຊຸດຂອງໃບທະບຽນຄອບຄົວຂອງຄູ່ສົມລົດຄົນເກົາຫຼີ, ໃບທະບຽນພື້ນຖານ, ໃບທະບຽນການແຕ່ງງານ(ລະອຽດ) ແລະໃບທະບຽນຜູ້ຢູ່ອາໄສ
  • ໃນກໍລະນີເກີດລູກໃນຊ່ວງແຕ່ງງານ, ຕ້ອງໄດ້ຍື່ນທະບຽນຄອບຄົວທີ່ມີຊື່ຂອງເດັກນໍາອີກ.

  • 2ການພິຈາລະນາສັນຊາດ
    • ຫຼັງຈາກ ຍື່ນເອກະສານຂໍສັນຊາດເກົາຫຼີ ໄປແລ້ວ, ທາງໜ່ວຍງານທີ່ກ່ຽວຂ້ອງ ຈະພິຈາລະນາເອກະສານ, ປະເມີນການເຂົ້າຮ່ວມກິດຈະກຳ ຕ່າງໆທາງສັງຄົມເກົາຫຼີ, ເອີ້ນມາສຳພາດ ແລະ ດຳເນີນການກວດກາສະຖານະພາບຕົວຈິງຂອງຜູ້ຮ້ອງຂໍສັນຊາດ.

    • ສຳລັບ ຄົນຕ່າງປະເທດ ທີ່ກຳລັງໃຊ້ຊີວິດຜົວເມຍ ກັບພົນລະເມືອງເກົາຫຼີນັ້ນ ແມ່ນຈະໄດ້ຮັບການຍົກເວັ້ນຂັ້ນຕອນການປະເມີນການເຂົ້າ ຮ່ວມກິດຈະກຳທາງສັງຄົມເກົາຫຼີ ໂດຍຈະໄດ້ສຳພາດຢ່າງດຽວ ເພື່ອປະເມີນຄວາມສາມາດດ້ານພາສາເກົາຫຼີ ແລະ ຄວາມຮູ້ພື້ນຖານ ທີ່ຄົນເກົາຫຼີ ຄວນຕ້ອງໄດ້ຮູ້ ແລະ ມີໄວ້.
      • ສຳລັບ ບຸກຄົນຕ່າງປະເທດ ທີ່ໄດ້ຍ່າຮ້າງກັບພົນລະເມືອງເກົາຫຼີແລ້ວນັ້ນ, ຈະບໍ່ໄດ້ຮັບການຍົກເວັ້ນຂັ້ນຕອນການປະເມີນ ການເຂົ້າ ຮ່ວມກິດຈະກຳທາງສັງຄົມ.
    • ຖ້າສຳພາດຄັ້ງທີ່1 ບໍ່ຜ່ານ, ແມ່ນຈະໃຫ້ໂອກາດ ໃຫ້ເຂົ້າສຳພາດຕຶ່ມອີກ 1 ຄັ້ງຕຶ່ມ(ໃຫ້ໂອກາດສຳພາດ 2 ຄັ້ງ).
  • 3ພິທີຮັບຄຳສາບານຂອງການເປັນພົນລະເມືອງ ແລະ ໄດ້ຮັບສັນຊາດ
    • ໃບຮັບຮອງແມ່ນອະນຸຍາດພາຍຫຼັງທີ່ການທົບທວນຄືນສໍາລັບການປ່ຽນ, ຕ້ອງມີສ່ວນຮ່ວມໃນໃບຢັ້ງຢືນສັນຊາດທີ່ໄດ້ຮັບພິທີ (ວັນທີຈະໄດ້ຮັບແຈ້ງລ່ວງຫນ້າໂດຍຫ້ອງກວດຄົນເຂົ້າເມືອງ ແລະ ຫ້ອງການລັດຖະບານຕ່າງປະເທດຂອງເຂດອໍານາດຕັດສິນທີ່ເໝາະສົມ. ເມື່ອໄດ້ຮັບການໃບຢັ້ງຢືນສັນຊາດຢູ່ໃນພິທີໃບຢັ້ງຢືນການໄດ້ຮັບສັນຊາດ ຫຼັງຈາກຄໍາສາບານວ່າເປັນສັນຊາດ, ສັນຊາດເກົາຫຼີທີ່ໄດ້ມາ. ເວລາທີ່ຕ້ອງການຂອງການທົບທວນຄືນການປ່ຽນສັນຊາດ ຈະຖືກປະກາດໃນແຕ່ລະເດືອນຜ່ານໜ້າເວັບທໍາອິດຂອງລັດຖະບານ (www.hikorea.go.kr), ການບໍລິການຄົນເຂົ້າເມືອງຂອງເກົາຫຼີ, ຫຼື ຂໍ້ມູນສັນຊາດທີ່ສົມບູນແບບຂອງ Kakao Talk Plus Friend.
    • ຫຼັງຈາກທີ່ໄດ້ຮັບໃບຢັ້ງຢືນການປ່ຽນສັນຊາດ, ຄວາມຖືກຕ້ອງຂອງຊື່ ແລະ ວັນເດືອນປີເກີດໃນໃບຢັ້ງຢືນຄວນໄດ້ຮັບການກວດກາ.
  • 4ການປະກາດສະຫຼະສັນຊາດເດີມ (ສະຖານທູດຂອງປະເທດບ້ານເກີດ) ຫຼື ໃຫ້ການສາບານວ່າ ຈະບໍ່ໃຊ້ສັນຊາດຕ່າງປະເທດອີກ (ອົງການກວດຄົນເຂົ້າເມືອງ ແລະ ບໍລິການຄົນຕ່າງປະເທດ(ສຳນັກງານໃຫຍ່))
    ຜູ້ຍື່ນຄຳຮ້ອງ ຕ້ອງສາບານໂຕ ຕໍ່ລັດຖະມົນຕີກະຊວງຍຸຕິທໍາວ່າ ຕົນຈະສະຫຼະສັນຊາດເດີມຂອງຕົນ ຫຼືຈະບໍ່ໃຊ້ສິດ ຊາວຕ່າງຊາດໃນເກົາຫຼີພາຍໃນເວລາ 1 ປີ ນັບຈາກວັນທີອະນຸມັດ ໄດ້ຮັບສັນຊາດເກົາຫຼີ ເປັນຕົ້ນໄປ
    • ຖ້າຜູ້ສະໝັກບໍ່ປະລະສັນຊາດຕ່າງປະເທດຂອງຕົນພາຍໃນໜຶ່ງປີຂອງການໄດ້ຮັບສັນຊາດເກົາຫຼີ ຫຼື ບໍ່ສາມາດຍື່ນສັນຍາກ່ຽວກັບການບໍ່ໃຊ້ສັນຊາດຕ່າງປະເທດ, ສັນຊາດເກົາຫຼີຈະຖືກຖອນຄືນ. ເພື່ອຮັບສັນຊາດເກົາຫຼີຄືນໃໝ່, ຜູ້ສະໝັກຕ້ອງລາຍງານການປະຕິເສດ ແລະ ສະໝັກຄືນໃໝ່ ເພື່ອຂໍສັນຊາດເກົາຫຼີ ຫຼື ການຟື້ນຟູສັນຊາດ.
    ຜູ້ສະໝັກທີ່ປະຕິເສດການເປັນພົນລະເມືອງຂອງເຂົາເຈົ້າ ຕ້ອງຍື່ນໃບຢັ້ງຢືນຢ່າງເປັນທາງການກ່ຽວກັບການປະຖິ້ມຊື່ທີ່ອອກໂດຍສະຖານທູດຂອງເຂົາເຈົ້າ (ສະຖານກົງສຸນ) ໄປທີ່ຫ້ອງການຄົນເຂົ້າເມືອງ ແລະ ຫ້ອງການລັດຖະບານຕ່າງປະເທດທີ່ຢູ່ໃນຂອບເຂດສິດອໍານາດ ແລະ ໄດ້ຮັບເອກະສານຢັ້ງຢືນ.
    • ກະລຸນາຕິດຕໍ່ຫາສະຖານທູດສໍາລັບເອກະສານທີ່ຈໍາເປັນ, ແລະ ອື່ນໆ, ສໍາລັບການຖອນສັນຊາດຕ່າງປະເທດ.
    ຜູ້ຍື່ນຕ້ອງສາບານວ່າຈະບໍ່ໃຊ້ສັນຊາດຕ່າງຊາດ ພຽງແຕ່ເມື່ອຜູ້ກ່ຽວຮັກສາຄວາມສຳພັນການແຕ່ງງານໄວ້, ແລະ ຂໍເອົາເອກະສານການຢືນຢັນສຳລັບການສາບານວ່າຈະບໍ່ໃຊ້ສັນຊາດຕ່າງຊາດທີ່ອອກໃຫ້ຫຼັງຈາກການສາບານ.

    • ສຳລັບປະເທດທີ່ບໍ່ອະນຸມັດໃຫ້ມີຫຼາຍສັນຊາດ (ຈີນ ແລະອື່ນໆ), ຜູ້ຍື່ນອາດຈະສູນເສຍສັນຊາດເດີມ ອີງຕາມກົດໝາຍຂອງປະເທດນັ້ນໆ. (ໃຫ້ສອບຖາມສະຖານທູດບ້ານເກີດສຳລັບການອະນຸຍາດໃຫ້ມີຫຼາຍສັນຊາດ).

    • ຖ້າຕ້ອງການ, ສາມາດສົ່ງບັດທະບຽນຄົນຕ່າງດ້າວໄດ້ລ່ວງໜ້າເມື່ອໄດ້ຮັບເອກະສານຢືນຢັນການສະຫຼະສັນຊາດຕ່າງຊາດ ຫຼືເອກະສານຢືນຢັນການບໍ່ໃຊ້ສັນຊາດຕ່າງຊາດແລ້ວ.

(2)ການອອກບັດປະຊາຊົນ(ບັດປະຈຳຕົວ)

ເມື່ອທ່ານຈະໄດ້ຮັບສັນຊາດເກົາຫຼີ ແລະ ໄດ້ແຈ້ງຂໍບັດປະຈຳຕົວພົນລະເມືອງແລ້ວ ໜ່ວຍງານທີ່ກ່ຽວຂ້ອງ ຈະອອກບັດປະຈຳຕົວໃຫ້. ບັດປະຈຳຕົວນີ້ ສະແດງໃຫ້ເຫັນວ່າ ເຮົາ ຕົນເອງໄດ້ກາຍເປັນພົນລະເມືອງເກົາຫຼີ ຢ່າງສົມບູນແລ້ວ. ບັດດັ່ງກ່າວນີ້ ຈະຕ້ອງໄດ້ສະແດງໃຫ້ເຈົ້າໜ້າທີ່ກ່ຽວຂ້ອງ ປະຈຳອົງການຈັດຕັ້ງຕ່າງ ເວລາເຮົາໄປສະເໜີ ຫຼື ຮ້ອງຂໍສິ່ງໃດໜຶ່ງ.

ຈົ່ງຢ່າໃຫ້ຄົນອື່ນຢືມບັດທະບຽນຜູ້ຢູ່ອາໄສຂອງທ່ານ ຫຼືບອກເລກທະບຽນຜູ້ຢູ່ອາໄສກັບຄົນອື່ນ. ຂໍ້ມູນການລົງທະບຽນຜູ້ຢູ່ອາໄສຂອງທ່ານອາດຈະຖືກໃຊ້ເພື່ອກໍ່ອາຊະຍາກໍາໄດ້.
  • ການອອກບັດທະບຽນຜູ້ຢູ່ອາໄສໃໝ່: ຖ້າທ່ານເຮັດບັດທະບຽນຜູ້ຢູ່ອາໄສເສຍ, ທ່ານອາດຈະໄປຂໍໃໝ່ໄດ້ທີ່ເຂດທີ່ທ່ານອາໃສ ເຊັ່ນ: ຫ້ອງການບ້ານ, ເມື່ອງ ຂອງທ່ານໂດຍກຽມຮູບຖ່າຍ (3x4ຊມ) ທີ່ຖ່າຍພາຍໃນເວລາຫົກເດືອນຜ່ານມາໄປນໍາ.
  • ໃນກໍລະນີມີການປ່ຽນບ່ອນຢູ່: ທ່ານຕ້ອງແຈ້ງທີ່ຢູ່ໃໝ່ຂອງທ່ານໃຫ້ເຂດທີ່ທ່ານອາໃສ ເຊັ່ນ: ຫ້ອງການບ້ານ, ເມື່ອງ ໃໝ່ຂອງທ່ານຮັບຮູ້ພາຍໃນ 14 ວັນ.
ຖ້າທ່ານບໍ່ແຈ້ງທີ່ຢູ່ໃໝ່ຂອງທ່ານ, ການຈົດທະທບຽນຜູ້ຢູ່ອາໄສຂອງທ່ານສາມາດຖືກຍົກເລີກຫຼັງຈາກມີການກວດພົບ.
