ເພື່ອສົ່ງເສີມການຮ່ວມມືລະຫວ່າງປະເທດ ແລະ ສົ່ງເສີມຜູ້ນໍາລະດັບໂລກໂດຍຜ່ານການແລກປ່ຽນຊາວໜຸ່ມ, ລັດຖະບານເກົາຫຼີໃຫ້ກິດຈະກໍາລະຫວ່າງປະເທດຕ່າງໆແກ່ຊາວໜຸ່ມ ແລະ ສົ່ງເສີມໃຫ້ຄອບຄົວຫຼາຍວັດທະນະທໍາເຂົ້າຮ່ວມໃນກິດຈະກໍາເຫຼົ່ານີ້.

01ແນະນຳໂຄງການແລກປ່ຽນລະຫວ່າງປະເທດ

ແນະນຳໂຄງການແລກປ່ຽນລະຫວ່າງປະເທດ
ໂຄງການ ຄຸນວຸດທິ ໄລຍະ ເນື້ອໃນ
ການແລກປ່ຽນຊາວໝຸ່ມສາກົນ ໄວໜຸ່ມອາຍຸ 16-24 10 ວັນ ປັບປຸງ​ຄວາມ​ສາມາດ​ຂອງ​ຊາວ​ໜຸ່ມ​ ເກົາຫຼີ ​ໃນ​ການ​ເຂົ້າ​ໃຈ​ວັດທະນະທຳ​ອື່ນໆ ​ໂດຍ​ຜ່ານ​ການ​ພົບ​ປະ​ແລກປ່ຽນ​ກັບ​ຊາວ​ໜຸ່ມ​ຈາກ​ບັນດາ​ປະ​ເທດ​ທີ່​ໄດ້​ລົງ​ນາມ​ໃນ​ຂໍ້​ຕົກລົງ, ການ​ຢ້ຽມຢາມ​ບັນດາ​ອົງການ​ຊາວ​ໜຸ່ມ ແລະ ການ​ເຄື່ອນ​ໄຫວ​ປະສົບ​ການ​ດ້ານ​ວັດທະນະທຳ.
ກຸ່ມອາສາສະໝັກຊາວ ໜຸ່ມຢູ່ຕ່າງປະເທດ “ລອຍເຂົ້າໄປໃນຄວາມຝັນ ແລະ ຄວາມຮັກ” ໄວໜຸ່ມອາຍຸ 15-24 ພາຍໃນ 9 ມື້
  • ແຕກຕ່າງໄປຕາມອົງການ
 ຊ່ວຍຮັບຮູ້ທ່າແຮງຂອງໄວໜຸ່ມ ແລະ ເສີມສ້າງຄວາມສາມາດຂອງເຂົາເຈົ້າໃນທົ່ວໂລກໂດຍຜ່ານກິດຈະກໍາອາສາສະໝັກຢູ່ຕ່າງປະເທດ
ກຸ່ມເຂົ້າຮ່ວມກອງປະຊຸມສາກົນ “ຊຸກຍູ້ໃຫ້ຊາວໜຸ່ມກາຍເປັນຜູ້ ນຳຂອງໂລກ” ໄວໜຸ່ມອາຍຸ 15-24 ປະມານ 3 ມື້
  • ແຕກຕ່າງກັນໄປຕາມກອງປະຊຸມ
 ເປັນຕົວແທນຂອງເກົາຫຼີ ຢູ່ໃນກອງປະຊຸມ ແລະ ສົນທິສັນຍາສາກົນ, ລວມທັງຄະນະກໍາມະການສະຫະປະຊາຊາດຄັ້ງທີ 3, ແລະ ໄດ້ມີການສົນທະນາແລະ ແລກປ່ຽນກັບຕົວແທນຊາວໜຸ່ມຈາກຕ່າງປະເທດ

02ສິດທິພິເສດ ສຳລັບຄອບຄົວຫຼາຍວັດທະນະທຳ

  • ເວລາຄັດເລືອກ ຈະໃຫ້ບຸລິມະສິດພິເສດ ແກ່ເຍົາວະຊົນ ທີ່ມາຈາກຄອບຄົວຫຼາຍວັດທະນະທຳ
    • ໃນກໍລະນີ ເປັນການແລກປ່ຽນເຍົາວະຊົນ ລະຫວ່າງ ປະເທດ, ຈະໃຫ້ສິດທິພິເສດ ສຳລັບຜູ້ທີ່ສະໝັກໄປປະເທດບ້ານເກີດ ຂອງພໍ່ ຫຼື ແມ່ຂອງເຂົາເຈົ້າ
  • ສິດທິພິເສດ
    • ຊ່ວຍເຫຼືອຄ່າໃຊ້ຈ່າຍຕ່າງໆໃນການເດີນທາງ ເພື່ອສົ່ງເສີມໃຫ້ເຍົາວະຊົນ ມີໂອກາດໄປແລກປ່ຽນກັບຕ່າງປະເທດ.

ສິ່ງທີ່ຄວນຮູ້ໄວ້!
ໂຄງການສົ່ງເສີມຄວາມກ້າວໜ້າຂອງເດັກລູກຊ໊ອດ

ໂຄງການສົ່ງເສີມຄວາມກ້າວໜ້າຂອງເດັກນ້ອຍ ຄອບຄົວຫຼາຍວັດທະນະທຳ ແມ່ນໂຄງການ ທີ່ໃຫ້ການຊ່ວຍເຫຼືອແກ່ບັນ ດາເຍົາວະຊົນ ທີ່ເປັນເດັກ ຄອບຄົວຫຼາຍວັດທະນະທຳ ເພື່ອໃຫ້ເຂົາເຈົ້າ ຮັບຮູ້ຕົນເອງ, ມີຄວາມເຂົ້າໃຈຕໍ່ສັງຄົມ, ສ້າງຄວາມເປັນຜູ້ນຳ, ວາງແຜນເສັ້ນທາງເດີນຂອງຊີວິດ ແລະ ອື່ນໆ ພ້ອມທັງສ້າງຄວາມປອງດອງກັນ ລະຫວ່າງພໍ່ແມ່ ແລະ ລູກ ໃນຄອບຄົວ. ນອກຈາກນີ້, ໂຄງການ ຍັງມີການຮ່ວມມືກັບບັນດາອົງການທີ່ກ່ຽວຂ້ອງ ພາຍໃນແຕ່ລະເຂດ ເພື່ອບໍລິການຊ່ວຍເຫຼືອທີ່ເປັນລະບົບ ແກ່ບັນດາເດັກລູກຊ໊ອດ ໃຫ້ມີຄວາມກ້າວໜ້າ ເພື່ອໃຫ້ເຂົາເຈົ້າ ປະກອບສ່ວນເຂົ້າໃນ ການສ້າງຄວາມເຂັ້ມແຂງໃຫ້ແກ່ປະເທດ, ໂດຍຢູ່ໃນສູນຊ່ວຍເຫຼືອຄອບຄົວຫຼາຍວັດທະນະທຳ ແມ່ນມີຫຼາຍກິດຈະການ ເພື່ອ ຊ່ວຍເຫຼືອແກ່ບັນດາຄອບຄົວຫຼາຍວັດທະນະທຳ ຢູ່ໃນເກົາຫຼີ.

ເປົ້າໝາຍຂອງໂຄງການ :
ລູກ ຈາກຄອບຄົວຫຼາຍວັດທະນະທຳ ທີ່ຢູ່ໃນເກນອາຍຸ, ພໍ່ແມ່ ແລະ ເຍົາວະຊົນ ທີ່ເຂົ້າປະເທດ ເກົາຫຼີ ໃນກາງປີ
ອົງການດຳເນີນງານ :
ອົງການທີ່ບໍລິການຊ່ວຍເຫຼືອແກ່ຄອບຄົວຫຼາຍວັດທະນະທຳ ຈຳນວນ 206 ແຫ່ງ.
ເນື້ອໃນຕົ້ນຕໍ
  • ໂຄງການເພື່ອສົ່ງເສີມຄວາມສໍາພັນກັນລະຫວ່າງພໍ່ແມ່ແລະເດັກນ້ອຍ: ການສ້າງຕັ້ງຕົວຕົນຂອງຕົນເອງ, ໃຫ້ຄໍາປຶກສາໃນຄອບຄົວ, ການສຶກສາພໍ່ແມ່ (ພາລະບົດບາດຂອງພໍ່ແມ່, ຄວາມສາມາດສື່ສານກັບເດັກນ້ອຍ, ການຄຸ້ມຄອງຂໍ້ຂັດແຍ່ງ, ແລະ ອື່ນໆ), ການຕັ້ງແຄ້ມ ຂອງຄອບຄົວ
  • ໂຄງການເພື່ອພັດທະນາທັກສະທາງດ້ານສັງຄົມ: ການບົ່ງມະຕິຂອງທັກສະທາງດ້ານສັງຄົມ, ສະໂມສອນຊາວຫນຸ່ມ (ກຸ່ມການຊ່ວຍເຫຼືອຕົນເອງ), ການເຮັດວຽກອາສາສະຫມັກ, ການຕັ້ງແຄ້ມເປັນຜູ້ນໍາ, ກິດຈະກໍາພາກສະຫນາມ, ແລະ ອື່ນໆ.
  • ໂຄງການການອອກແບບອະນາຄົດ: ການສ້າງແຮງຈຸງໃຈ, ການທົດສອບຄວາມຖະຫນັດ, ການສຶກສາເສັ້ນທາງການປະກອບອາຊີບ, ການອອກແບບອາຊີບ, ກິດຈະກໍາປະສົບການໆເຮັດວຽກ, ແລະ ອື່ນໆ.
  • ໂຄງການສໍາລັບການສະຫນັບສະຫນູນເພື່ອເອົາຊະນະວິກິດການ: ໃຫ້ຄໍາປຶກສາແກ່ຊາວຫນຸ່ມ, ພໍ່ແມ່, ແລະ ບຸກຄົນ, ການປິ່ນ ປົວທາງຈິດໃຈ ແລະ ທາງດ້ານອາລົມຈິດ, ສິນລະປະ/ດົນຕີ/ການປິ່ນປົວສໍາລັບການຫຼິ້ນ
  • ເຄືອຂ່າຍຊຸມຊົນພູມິພາກ: ເຊື່ອມຕໍ່ກັບສະຖາບັນທີ່ກ່ຽວຂ້ອງເຊັ່ນ ສູນໃຫ້ຄໍາປຶກສາຊາວຫນຸ່ມ, ສູນສະຫວັດດີການ, ໂຮງຮຽນສາຍຮຸ້ງ, ສູນຝຶກອົບຮົມຊາວຫນຸ່ມ, ສູນກາງສະຫນັບສະຫນູນຊາວຫນຸ່ມຢູ່ນອກໂຮງຮຽນ, ແລະ ອື່ນໆ.
